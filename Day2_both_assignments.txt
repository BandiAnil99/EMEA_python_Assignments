Q1. Student Marks Analyzer


User Story:

A teacher wants to analyze the marks of students and identify students who scored above 75.


Given:


students = [

    ("Anu", 85),

    ("Ravi", 62),

    ("Maya", 91),

    ("John", 74),

    ("Sara", 88)

]


Task:

Use list comprehension and tuples to create a list containing the names of students who scored above 75.


Answer:


students = [

    ("Anu", 85),

    ("Ravi", 62),

    ("Maya", 91),

    ("John", 74),

    ("Sara", 88)

]


names = [name for (name, marks) in students if marks > 75]

print(names)


Output:

['Anu', 'Maya', 'Sara']

2. Deposit

3. Withdraw

4. Exit


Answer:

for i in range(2):

    key = input("Enter key:")

    value = input("Enter value:")

    account[key] = value


def check_balance(account):

    print(account["balance"])


def deposit(account, amnt):

    account["balance"] = int(account["balance"]) + amnt

    print(account["balance"])


def withdraw(account, amnt):


    if amnt > account["balance"\]:

        print("Amount exceeds balance")


    else:

        account["balance"] -= amnt

        print("Balance:", account["balance"])


while True:


    print("1.Check Balance")

    print("2.Deposit")

    print("3.Withdraw")

    print("4.Exit")


    opt = int(input("Enter option:"))


    if opt == 1:

        check_balance(account)


    elif opt == 2:

        amnt = int(input("Enter amount:"))

        deposit(account, amnt)


    elif opt == 3:

        amnt = int(input("Enter amount:"))

        withdraw(account, amnt)

Q4. Restaurant Order System


User Story:

Allow customer to order multiple items.


Answer:


menu = {

    "pizza": 250,

    "burger": 150,

    "pasta": 200,

    "juice": 80

}


ordered_items = []


def final_price(ordered_items):


    total = 0


    for i in ordered_items:

        total += menu[i]


    if total > 1000:

        total = total - total * 0.15

        print("Eligible for 15% discount")


    print("Final Price:", total)


while True:


    print("Menu")


    for item, price in menu.items():

        print(item, "-", price)


    item = input("Enter item:").lower()


    if item in menu:

        ordered_items.append(item)


    else:

        print("Item not available")


    ch = input("Order more? y/n:")


    if ch == 'n':

        final_price(ordered_items)

        break

Q6. Patient Temperature Monitor


User Story:

Monitor temperature of 5 patients.


Answer:


def classification(temp):


    if temp < 37:

        return "Normal"


    elif temp <= 38:

        return "Mild Fever"


    else:

        return "High Fever"


hsp = {}

cnt = 0


for i in range(5):


    name = input("Enter patient name:")

    temp = int(input("Enter temperature:"))


    hsp[name] = temp


for name, temp in hsp.items():


    status = classification(temp)


    print(name, ":", status)


    if status != "Normal":

        cnt += 1


print("FEVERED Patients:", cnt)

Q7. Library Book Tracker


User Story:

Track overdue books.


Answer:


library = []


def overdue_books(bname, name, days):


    if days > 7:

        return "Overdue"


    return "Not Overdue"


while True:


    bname = input("Book name:")

    name = input("Person name:")

    days = int(input("Borrowed days:"))


    library.append((bname, name, days))


    ch = input("Add more? y/n:")


    if ch == 'n':

        break


count = 0


for bname, name, days in library:


    if overdue_books(bname, name, days) == "Overdue":


        print(bname, "-", name, "-", days)


        count += 1


print("Overdue books:", count)

free = 0

child = 0

adult = 0

senior = 0

    if age >= 60:

        return 120


    elif age >= 13:

        return 200

        return 100


    else:

        return 0


for i in range(num):

        child += 1


    elif age <= 59:

Q10. Employee Salary Analyzer


User Story:

Answer:


employees = {

    "Anu": 45000,

    "Rahul": 65000,

    "Meera": 55000,

    "Arun": 80000

}


def highest_salary(emp):


    name = max(emp, key=emp.get)


    print("Highest Paid Employee:", name, emp[name])


def average_salary(emp):


    avg = sum(emp.values()) / len(emp)


    print("Average Salary:", avg)


def bonus_calculation(emp):


    print("Employees With Bonus:")


    for name, salary in emp.items():


        if salary < 50000:

            print(name, ":", salary + salary * 0.10)


print("Employee Details:")


for name, salary in employees.items():

    print(name, ":", salary)


highest_salary(employees)


low = min(employees, key=employees.get)


print("Lowest Paid Employee:", low, employees[low])


average_salary(employees)


print("Employees Earning More Than 60000:")


for name, salary in employees.items():


   if salary > 60000:

        print(name, ":", salary)
